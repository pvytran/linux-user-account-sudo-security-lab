# Linux User Account & Sudo Security Lab

# Verify root
whoami

# Create omega3
useradd -c "Cocoa" -s /bin/bash -m omega3
passwd omega3

# Create standard user
useradd -c "Jumbo" -s /bin/bash -m sunflower
passwd sunflower

# Verify accounts
id omega3
id sunflower

# Install sudo
apt update
apt install sudo

# Check sudo group
grep '^sudo:' /etc/group

# Give omega3 sudo access
usermod -aG sudo omega3

# Test sudo
whoami
sudo whoami

# Locate nologin
find / -type f -name nologin 2>/dev/null

# Disable interactive root login
sudo /usr/sbin/usermod -s /usr/sbin/nologin root

# Verify root configuration
grep '^root:' /etc/passwd

# Test root login restriction
su - root

# Final verification
id omega3
id sunflower
grep '^sudo:' /etc/group
grep '^root:' /etc/passwd
